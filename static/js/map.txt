function updateFieldMapPixi_6(newPoints) {
    if (!newPoints || newPoints.length === 0) return;

    // Добавляем новые точки в очередь
    pointsQueue.push(...newPoints);

    if (isProcessingQueue) return;
    isProcessingQueue = true;

    const CHUNK_SIZE = 1000;

    function processNextChunk() {
        if (pointsQueue.length === 0) {
            isProcessingQueue = false;
            lastQueueProgressTime = performance.now(); // Сторож спокоен
            return;
        }

        const chunk = pointsQueue.splice(0, CHUNK_SIZE);

        chunk.forEach((pt) => {
            if (!pt || pt.length < 5) return;

            // Корректно распаковываем массив с учетом ключа чанка от сервера
            const isNewFormat = typeof pt[0] === 'string';
            const lat = isNewFormat ? pt[1] : pt[0];
            const lon = isNewFormat ? pt[2] : pt[1];
            const hdg = isNewFormat ? pt[3] : pt[2];
            const states = isNewFormat ? pt[4] : pt[3];
            const customWidths = (isNewFormat ? pt[5] : pt[4]) || null;

            // 1. ОПРЕДЕЛЯЕМ ЧАНК НА ОСНОВЕ ТЕКУЩЕЙ ПОЗИЦИИ КАРТЫ (targetX, targetY)
            // Мы берем твои реальные пиксели поля, к которым плавно едет трактор.
            // Делим их на 4000 (размер холста RenderTexture).
            const cx = Math.floor(targetX / 4000);
            const cy = Math.floor(targetY / 4000);
            const chunkKey = `${cx}_${cy}`;

            // 2. МЕНЕДЖЕР ЧАНКОВ: Если такого квадрата 4000х4000 еще нет — создаем его
            if (!mapChunks[chunkKey]) {
                console.log(`[FRONTEND CHUNKS] Создаем холст RenderTexture для локального чанка: ${chunkKey}`);

                mapChunks[chunkKey] = PIXI.RenderTexture.create({ width: 4000, height: 4000 });
                const chunkSprite = new PIXI.Sprite(mapChunks[chunkKey]);

                // Размещаем спрайт чанка ровно по пиксельной сетке поля
                chunkSprite.x = cx * 4000;
                chunkSprite.y = cy * 4000;

                // Добавляем чанк в world, чтобы он двигался и крутился вместе со всем полем!
                world.addChild(chunkSprite);
            }

            // 3. Очищаем графику перед отрисовкой сегмента штанги
            tempTrackGraphics.clear();

            // 4. Твоя родная тригонометрия штанги малює полигон в tempTrackGraphics
            appendTrackSegmentPixi(lat, lon, hdg, states, customWidths);

            // 5. ЗАПЕКАЕМ СЕГМЕНТ В ТЕКСТУРУ С УЧЕТОМ СДВИГА МАТРИЦЫ CHANKA
            if (mapChunks[chunkKey]) {
                const renderMatrix = new PIXI.Matrix();
                // Сдвигаем матрицу рисования в локальные координаты текущей текстуры чанка
                renderMatrix.translate(-(cx * 4000), -(cy * 4000));

                app.renderer.render(tempTrackGraphics, {
                    renderTexture: mapChunks[chunkKey],
                    clear: false,
                    matrix: renderMatrix
                });
            }
        });

        // Полностью очищаем векторную память GPU после прохода пачки
        tempTrackGraphics.clear();

        // Обновляем счетчик полигонов на UI
        const polyElem = document.getElementById('total_point');
        if (polyElem) {
            polyElem.innerText = "Poligon count: " + totalPolygonsRendered;
        }

        // Переходим к следующей пачке
        setTimeout(processNextChunk, 1);
    }

    processNextChunk();
}

        // Будемо групувати точки з пачки за їхніми чанками, щоб правильно запікати в GPU
        let currentChunkKey = null;

        chunk.forEach((pt) => {
            // Перевіряємо новий формат від Python: [chunk_key, lat, lon, hdg, states, widths]

            const widths = pt[5] || null;
            const customWidths = pt[5] || null;
